基于Docker Compose的Kafka分布式集群部署

什么是Kafka

kafka是一个分布式流处理平台，主要用于构建实时数据管道和流应用。它具备高吞吐，低延迟，可拓展和容错等特性。
主要功能包括：

  • 发布和订阅数据流：类似消息队列，支持多生产者和多消费者。
  • 存储数据流：数据可以被持久化
  • 实时处理数据流：支持对数据进行转换，聚合等实时处理操作。
    应用场景：
  • 日志收集与聚合
  • 事件溯源
  • 消息队列服务

搭建的最新版kafka集群

本次搭建kafka集群使用的是docker-compose来进行搭建所需要的环境。docker-compose文件如下所示：

version: '3.8'
services:
  kafka1:
    image: apache/kafka:latest
    container_name: kafka1
    ports:
      - "9092:9092"
    environment:
      # Kafka 节点唯一标识符，每个节点必须不同
      KAFKA_NODE_ID: 1
      # 当前节点的角色：broker 和 controller
      KAFKA_PROCESS_ROLES: broker,controller
      # 监听器配置：PLAINTEXT 表示明文通信，CONTROLLER 用于 controller 间通信
      KAFKA_LISTENERS: PLAINTEXT://0.0.0.0:9092,CONTROLLER://0.0.0.0:9093
      # 外部可访问的监听地址（外部连接使用），IP 替换为宿主机实际 IP
      KAFKA_ADVERTISED_LISTENERS: PLAINTEXT://192.168.0.105:9092
      # 指定 controller 使用的监听名称
      KAFKA_CONTROLLER_LISTENER_NAMES: CONTROLLER
      # 协议映射：将监听器名称映射到协议类型
      KAFKA_LISTENER_SECURITY_PROTOCOL_MAP: CONTROLLER:PLAINTEXT,PLAINTEXT:PLAINTEXT
      # Controller Quorum 配置，指定参与投票的 controller 节点及端口
      KAFKA_CONTROLLER_QUORUM_VOTERS: 1@kafka1:9093,2@kafka2:9093,3@kafka3:9093
      # 日志目录位置
      KAFKA_LOG_DIRS: /kafka-logs
      # 偏移量主题副本因子，建议与集群节点数一致以提高容错能力
      KAFKA_OFFSETS_TOPIC_REPLICATION_FACTOR: 3
      # 事务状态日志副本因子
      KAFKA_TRANSACTION_STATE_LOG_REPLICATION_FACTOR: 3
      # 事务状态日志最小 ISR（In-Sync Replicas）数量
      KAFKA_TRANSACTION_STATE_LOG_MIN_ISR: 2
      # 默认分区数量，适用于新建的主题
      KAFKA_NUM_PARTITIONS: 3

  kafka2:
    image: apache/kafka:latest
    container_name: kafka2
    ports:
      - "9094:9092"
    environment:
      KAFKA_NODE_ID: 2
      KAFKA_PROCESS_ROLES: broker,controller
      KAFKA_LISTENERS: PLAINTEXT://0.0.0.0:9092,CONTROLLER://0.0.0.0:9093
      #ip为宿主机所在ip
      KAFKA_ADVERTISED_LISTENERS: PLAINTEXT://192.168.0.105:9094
      KAFKA_CONTROLLER_LISTENER_NAMES: CONTROLLER
      KAFKA_LISTENER_SECURITY_PROTOCOL_MAP: CONTROLLER:PLAINTEXT,PLAINTEXT:PLAINTEXT
      KAFKA_CONTROLLER_QUORUM_VOTERS: 1@kafka1:9093,2@kafka2:9093,3@kafka3:9093
      KAFKA_LOG_DIRS: /kafka-logs
      KAFKA_OFFSETS_TOPIC_REPLICATION_FACTOR: 3
      KAFKA_TRANSACTION_STATE_LOG_REPLICATION_FACTOR: 3
      KAFKA_TRANSACTION_STATE_LOG_MIN_ISR: 2
      KAFKA_NUM_PARTITIONS: 3
    volumes:
      - ./kafka2-data:/kafka-logs
    networks:
      - kafka-net

  kafka3:
    image: apache/kafka:latest
    container_name: kafka3
    ports:
      - "9096:9092"
    environment:
      KAFKA_NODE_ID: 3
      KAFKA_PROCESS_ROLES: broker,controller
      KAFKA_LISTENERS: PLAINTEXT://0.0.0.0:9092,CONTROLLER://0.0.0.0:9093
      #ip为宿主机所在ip
      KAFKA_ADVERTISED_LISTENERS: PLAINTEXT://192.168.0.105:9096
      KAFKA_CONTROLLER_LISTENER_NAMES: CONTROLLER
      KAFKA_LISTENER_SECURITY_PROTOCOL_MAP: CONTROLLER:PLAINTEXT,PLAINTEXT:PLAINTEXT
      KAFKA_CONTROLLER_QUORUM_VOTERS: 1@kafka1:9093,2@kafka2:9093,3@kafka3:9093
      KAFKA_LOG_DIRS: /kafka-logs
      KAFKA_OFFSETS_TOPIC_REPLICATION_FACTOR: 3
      KAFKA_TRANSACTION_STATE_LOG_REPLICATION_FACTOR: 3
      KAFKA_TRANSACTION_STATE_LOG_MIN_ISR: 2
      KAFKA_NUM_PARTITIONS: 3
    volumes:
      - ./kafka3-data:/kafka-logs
    networks:
      - kafka-net

networks:
  kafka-net:
    driver: bridge

环境配置参数说明参考：https://hub.docker.com/r/bitnami/kafka#environment-variables

创建docker-compose.yaml文件并且将以上内容复制，将KAFKA_ADVERTISED_LISTENERS 环境变量的ip替换为自己所在宿主机的ip，替换完成后使用以下命令进行启动kafka集群

docker-compose up -d

启动效果为：

使用 docker-compose ps kafka集群即为启动成功。通过宿主机ip:port即可访问到对应的kafka集群。
