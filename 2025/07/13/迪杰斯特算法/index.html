public PathResult dijkstra(int start, int end) {
    // 初始化记录起点到各顶点的距离
    double[] dist = new double[count];
    // 记录最短路径的前驱节点
    int[] path = new int[count];
    // 标记节点是否已处理
    boolean[] visited = new boolean[count];

    // 初始化邻接矩阵和距离数组
    for (int i = 0; i < count; i++) {
        dist[i] = adjMatrix[start][i];
        path[i] = dist[i] == Double.MAX_VALUE ? -1 : start;
    }

    // 标记起点为已访问
    visited[start] = true;
    dist[start] = 0;

    // 主循环：寻找最短路径
    for (int i = 0; i < count; i++) {
        // 找到当前未访问的最近节点
        int index = start;
        double min = Double.MAX_VALUE;
        for (int j = 0; j < count; j++) {
            if (!visited[j] && dist[j] < min) {
                index = j;
                min = dist[j];
            }
        }

        // 标记该节点为已访问
        visited[index] = true;

        // 更新与当前节点相邻节点的最短距离
        for (int k = 0; k < count; k++) {
            double temp = adjMatrix[index][k] == Double.MAX_VALUE ? Double.MAX_VALUE : min;
            if (!visited[k] && dist[k] >= temp) {
                dist[k] = temp;
                path[k] = index;
            }
        }

        // 如果已处理终点，提前结束
        if (index == end) {
            break;
        }
    }

    // 构建实际路径
    List<VertexModel> result = new ArrayList<>();
    realPath(start, end, path, result);
    result.add(vertex.get(end));

    // 返回结果
    return new PathResult(dist[end], result);
}
